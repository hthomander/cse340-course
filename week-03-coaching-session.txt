Let’s start with models. In projects.js, getProjectDetails(id) selects a project and its organization name, uses $1 in the SQL, and passes id as the query parameter.

```js
const getProjectDetails = async (id) => {
  const query = `
    SELECT
      projects.id AS project_id,
      projects.title,
      projects.description,
      projects.date,
      projects.location,
      projects.organization_id,
      organization.name AS organization_name
    FROM public.projects AS projects
    JOIN public.organization AS organization
      ON projects.organization_id = organization.organization_id
    WHERE projects.id = $1;
  `;

  const result = await db.query(query, [id]);
  return result.rows[0] || null;
};
```
